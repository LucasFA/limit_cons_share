# All values are evaluated in country scope.

# How many construction points per week we assume each queued private project can absorb,
# as a multiple of Max Weekly Construction Progress. Raise this if private construction
# appears throttled while the cap is active; lower it if the government exceeds the cap.
lcs_private_demand_per_project_factor = 1

# The configured cap as a percentage (0-100). 100 = no cap.
lcs_cap_percent = {
	value = 100
	if = {
		limit = { has_variable = lcs_cap }
		value = var:lcs_cap
	}
}

# Construction points currently held back (display only; set by lcs_update_construction_cap)
lcs_withheld_points = {
	value = 0
	if = {
		limit = { has_variable = lcs_withheld }
		value = var:lcs_withheld
	}
}

# Construction sector throughput currently removed by lcs_construction_throttle
lcs_throttle_applied = {
	value = 0
	if = {
		limit = { has_variable = lcs_throttle }
		value = var:lcs_throttle
	}
}

# Construction sector throughput factor without this mod: 1 + country-level throughput bonuses.
# (Per-building/state throughput effects are not visible here; the weekly update corrects for them.)
lcs_nominal_throughput_factor = {
	value = 1
	add = modifier:building_construction_sector_throughput_add
	add = lcs_throttle_applied
	min = 0.01
}

# Construction sector throughput factor right now, including this mod's throttle
lcs_current_throughput_factor = {
	value = lcs_nominal_throughput_factor
	subtract = lcs_throttle_applied
	min = 0.01
}

# Private allocation currently added by lcs_private_construction_reserved
lcs_reserved_allocation = {
	value = 0
	if = {
		limit = { has_variable = lcs_reserved }
		value = var:lcs_reserved
	}
}

# Lowest construction sector throughput factor the mod will set, as a fraction of nominal.
# Keeps a little sector output so nominal construction can still be inferred from it.
lcs_min_throughput_fraction = 0.05

# Construction not produced by Construction Sectors, which throughput does not affect.
# Approximated by the GDP-based static modifier country_gdp_construction (country_construction_add = 10,
# scaled by GDP as described in 00_code_static_modifiers.txt). Traits etc. are ignored.
lcs_non_sector_construction = {
	value = gdp
	divide = define:NEconomy|COUNTRY_GDP_CONSTRUCTION_MODIFIER_DIVISOR
	multiply = define:NEconomy|COUNTRY_GDP_CONSTRUCTION_MODIFIER_SCALED_MULTIPLIER
	add = define:NEconomy|COUNTRY_GDP_CONSTRUCTION_MODIFIER_BASE_MULTIPLIER
	max = define:NEconomy|COUNTRY_GDP_CONSTRUCTION_MODIFIER_MAX_MULTIPLIER
	min = 0
	multiply = 10
	max = modifier:country_construction_add
}

# Construction currently produced by Construction Sectors (after the throttle)
lcs_current_sector_construction = {
	value = modifier:country_construction_add
	subtract = lcs_non_sector_construction
	min = 0
}

# Construction the country would produce without this mod. Throughput scales construction
# sector output linearly, so undo the throttle by the ratio of the throughput factors.
lcs_nominal_construction = {
	value = lcs_current_sector_construction
	multiply = lcs_nominal_throughput_factor
	divide = lcs_current_throughput_factor
	add = lcs_non_sector_construction
	min = 0
}

# Private construction allocation the country would have without this mod
lcs_nominal_private_allocation = {
	value = modifier:country_private_construction_allocation_mult
	subtract = lcs_reserved_allocation
	min = 0
	max = 1
}

# Rough estimate of how much construction the private queue can absorb this week:
# every queued private project can take at most Max Weekly Construction Progress.
lcs_private_demand_estimate = {
	value = num_queued_private_constructions
	multiply = modifier:country_max_weekly_construction_progress_add
	multiply = lcs_private_demand_per_project_factor
	min = 0
}

### The values below are only valid inside lcs_update_construction_cap, which sets
### lcs_base (nominal construction) and lcs_base_alloc (nominal private allocation).

# Most construction the government may use: cap% of nominal construction
lcs_government_limit = {
	value = var:lcs_base
	multiply = lcs_cap_percent
	divide = 100
}

# What the government is guaranteed in vanilla when private construction uses its full
# allocation, but no more than the cap
lcs_government_floor = {
	value = 1
	subtract = var:lcs_base_alloc
	multiply = var:lcs_base
	max = lcs_government_limit
	min = 0
}

# Construction set aside for the private sector: its estimated demand, but never so much
# that the government drops below its floor
lcs_private_reservation = {
	value = var:lcs_base
	subtract = lcs_government_floor
	max = lcs_private_demand_estimate
	min = 0
}

# Total construction to leave active: government limit + private reservation
lcs_target_construction = {
	value = lcs_government_limit
	add = lcs_private_reservation
	max = var:lcs_base
	min = 0
}

# Private allocation needed so private construction gets its reservation first
lcs_target_private_allocation = {
	value = var:lcs_base_alloc
	if = {
		limit = { lcs_target_construction > 0 }
		value = lcs_private_reservation
		divide = lcs_target_construction
		max = 1
		min = var:lcs_base_alloc
	}
}

# Construction sector throughput to remove so that total construction drops to
# lcs_target_construction. Sector output scales linearly with the throughput factor:
#   new factor = current factor * (target - non-sector) / current sector construction
# Starting from the current state means that if construction is already on target the throttle
# stays put, so throughput the model cannot see (per-building or state bonuses) is corrected
# over the following weeks.
lcs_target_throttle = {
	value = 0
	if = {
		limit = { var:lcs_cur_sector > 0 }
		value = var:lcs_nom_factor
		subtract = {
			value = lcs_target_construction
			subtract = var:lcs_non_sector
			min = 0
			multiply = var:lcs_cur_factor
			divide = var:lcs_cur_sector
		}
		min = 0
		max = {
			value = 1
			subtract = lcs_min_throughput_fraction
			multiply = var:lcs_nom_factor
		}
	}
}

### Display only (safe to evaluate at any time)

lcs_display_government_limit = {
	value = lcs_nominal_construction
	multiply = lcs_cap_percent
	divide = 100
}
